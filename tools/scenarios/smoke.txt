# smoke -- a self-test that exercises the core harness commands (and a visual
# baseline) so CI catches harness rot. Run:
#   tools/arbiter.sh smoke --headless --check-captures
# First time (to create the baseline): add --update-baselines.

seed 1

assert pc.hp > 0
assert pc.gold >= 0

wait 30
# visual baseline (full frame incl HUD) -> gated by --check-captures
capturescreen /tmp/arbiter-captures/town.bmp
# perf gates
assert-render-under 60 60
